import tkinter
import customtkinter
from pytubefix import YouTube
import imageio_ffmpeg
import subprocess
import os


def startdownload():
    try:
        # Get YouTube link
        ytLink = link.get().strip()

        finish.configure(
            text="Getting video information...",
            text_color="white"
        )

        # Create YouTube object
        ytObject = YouTube(
            ytLink,
            on_progress_callback=on_progress
        )

        # Display video title
        label.configure(text=ytObject.title)

        # Get highest quality MP4 video
        video = ytObject.streams.filter(
            only_video=True,
            file_extension="mp4"
        ).order_by("resolution").desc().first()

        # Get highest quality audio
        audio = ytObject.streams.filter(
            only_audio=True
        ).order_by("abr").desc().first()

        if video is None:
            finish.configure(
                text="No video stream found.",
                text_color="red"
            )
            return

        if audio is None:
            finish.configure(
                text="No audio stream found.",
                text_color="red"
            )
            return

        # Create temporary filenames
        video_file = "temp_video.mp4"
        audio_file = "temp_audio.m4a"

        finish.configure(
            text="Downloading video...",
            text_color="white"
        )

        # Download video
        video.download(
            filename=video_file
        )

        finish.configure(
            text="Downloading audio...",
            text_color="white"
        )

        # Download audio
        audio.download(
            filename=audio_file
        )

        finish.configure(
            text="Merging video and audio...",
            text_color="white"
        )

        # Get FFmpeg location
        ffmpeg = imageio_ffmpeg.get_ffmpeg_exe()

        # Clean the video title so it can be used as filename
        safe_title = "".join(
            c for c in ytObject.title
            if c.isalnum() or c in " ._-"
        ).strip()

        output_file = safe_title + ".mp4"

        # Merge video and audio
        subprocess.run(
            [
                ffmpeg,
                "-y",
                "-i", video_file,
                "-i", audio_file,
                "-c:v", "copy",
                "-c:a", "aac",
                output_file
            ],
            check=True
        )

        # Delete temporary files
        os.remove(video_file)
        os.remove(audio_file)

        # Set progress to 100%
        progressBar.set(1)
        percentage.configure(text="100%")

        finish.configure(
            text="Download complete!",
            text_color="green"
        )

    except Exception as e:
        print("ERROR:", e)

        finish.configure(
            text=f"Error: {e}",
            text_color="red"
        )


def on_progress(stream, chunk, bytes_remaining):

    totalSize = stream.filesize

    bytesDownloaded = totalSize - bytes_remaining

    percentage_of_completion = (
        bytesDownloaded / totalSize
    ) * 100

    per = str(int(percentage_of_completion))

    percentage.configure(
        text=per + "%"
    )

    percentage.update()

    progressBar.set(
        float(percentage_of_completion) / 100
    )


# -------------------------
# GUI
# -------------------------

customtkinter.set_appearance_mode("System")
customtkinter.set_default_color_theme("blue")

window = customtkinter.CTk()

window.geometry("720x480")
window.title("Youtube Downloader")


# Title
label = customtkinter.CTkLabel(
    window,
    text="Welcome to my YouTube Downloader"
)

label.pack(
    padx=10,
    pady=20
)


# Link textbox
var_link = tkinter.StringVar()

link = customtkinter.CTkEntry(
    window,
    height=40,
    width=400,
    textvariable=var_link
)

link.pack()


# Percentage
percentage = customtkinter.CTkLabel(window,text="0%")

percentage.pack(pady=10)


# Progress bar
progressBar = customtkinter.CTkProgressBar(window,width=400)

progressBar.set(0)

progressBar.pack(padx=10,pady=10)


# Status
finish = customtkinter.CTkLabel(window,text="")

finish.pack(pady=10)


# Download button
button = customtkinter.CTkButton(window, text="Download",command=startdownload)

button.pack(padx=10,pady=10)


window.mainloop()
